## Uppgift 8 (8p)

En elektroniskt styrd ventilationslucka ska konstrueras. Luckan har ett 7-bitars digitalt gränssnitt (se figur nedan.)

<p align="center">
  <img src="../images/ventilation.png" width="50%"/>
</p>


**Operatörsignaler:**
- En funktion för att ge larm via en högtalare aktiveras med en etta på bit 2.
- Funktioner för att öppna och stänga luckan utgörs av två återfjädrande strömbrytare anslutna via b1 och b0. Då strömbrytarna är öppna är dessa ingångar flytande och måste därför förses med programmerad pull-up.
- Om båda strömbrytarna aktiveras samtidigt ska larmsignal ges.

**Signaler till/från ventilationslucka:**
- Indikatorer för helt öppen respektive helt stängd lucka är
anslutna via b4 och b3. Signalerna är aktiva vid hög nivå.
Om båda dessa signaler är 0 betyder det att luckan håller på
att öppnas eller stängas. Om båda signaler är 1 innebär detta
något fel och larmet ska då aktiveras.
- Signalerna b6 och b5 aktiveras (sätts till 1) för att öppna
respektive stänga luckan. Dessa signaler får inte aktiveras
samtidigt. I mekaniken som reglerar luckan finns en viss
tröghet så det kan ta maximalt 1 sekund att öppna och
maximalt 2 sekunder att stänga luckan. Om det skulle ta
längre tid ska larmsignal ges.

För att kunna kontrollera att det inte tar mer än 1 sekund att öppna eller stänga luckan krävs en funktion för realtidsfördröjning användas. Du kan förutsätta att en sådan blockerande funktion `void delay_100ms(void);` som fördröjer det anropande programmet i 1/10 sekund finns tillgänglig.
Du kan alltså använda denna men behöver inte visa funktionen här.

Förutsätt att makrodefinitioner för pekare till använda register finns givna med namnkonventioner
enligt QuickGuide.

Konstruera tre funktioner, init, open och close enligt följande specifikationer:

a) `void init ( void );` initiera GPIOD för användning med luckan.
Oanvända pinnar i porten ska ställas som ingångar. (2p)

b) `int open ( void );` öppna luckan, vänta maximalt 1 sekund i 100 ms intervall.
Om dörren öppnas inom 1 sekund ska funktionen returnera 1,
annars ska funktionen returnera 0. (3p)

c) `int close (void);` stäng lucka, vänta maximalt 2 sekunder i 100 ms intervall.
Om dörren är stängd inom 2 sekunder ska funktionen returnera 1, annars ska funktionen returnera 0. (3p)

```asm
@ a0: cp
@ a1: converted
convert:
    li      a1, 0              @ converted = 0
L1:
    lbu     t0, 0(a0)          @ t0 <= *cp
    beq     t0, zero, L3        @ *cp == 0 ?
    sext.b  t0, t0             @ t0 <- (int8_t)*cp
    bge     t0, zero, L2        @ (int)*cp < 0 ?
    neg     t0, t0             @ t0 <- -(*cp)
    sb      t0, 0(a0)          @ *cp = -(*cp)
    addi    a1, a1, 1           @ converted++
L2:
    addi    a0, a0, 1           @ cp++
    j       L1
L3:
    mv      a0, a1             @ return converted
    ret
```

## Uppgift 5 (5p)
```c
/* TIM7 interrupt number (CH32V307): 71 */
#define TIM7_IRQn            71
#define PFIC_TIM7_IENR3_BIT  (1U << (TIM7_IRQn - 64))  /* bit 7 i IENR3 */

#define UIE (1 << 0)
#define CEN (1 << 0)

/* Exempel på vektorpost i RAM-baserad vektortabell */
#define TIM7_IRQVEC   ((volatile unsigned long *)(0x20000000UL + 4UL*TIM7_IRQn))

void timer7_init(void)
{
  *TIM7_IRQVEC = (unsigned long)timer7_interrupt;

  /* Enable TIM7 interrupt i PFIC */
  *PFIC_IENR3 |= PFIC_TIM7_IENR3_BIT;

  /* Systemklocka: 144 MHz 
   * Tick-frekvens: 144 MHz / 144,000 = 1 kHz (1 ms per tick)
   * Interrupt varje 100 ms (100 ticks × 1 ms/tick)
  */
  *TIM7_PSC   = 144 * 1000 - 1;
  *TIM7_ATRLR = 100 - 1;

  /* Update interrupt enable */
  *TIM7_DMAINTENR |= UIE;

  /* Start timer */
```

```c
#define PFIC_EXTI8_IENR1_BIT   (1 << 23)   /* exempel: EXTI8 i PFIC IENR1 */
#define EXTI8_BIT              (1 << 8)

void app_init(void)
{
  /* Port C, bit 8 utgång (push-pull) */
  /* GPIOC_CFGHR nibble för PC8 (bits 3:0): MODE=01, CNF=00 */
  *GPIOC_CFGHR &= ~0x0000000F;
  *GPIOC_CFGHR |= 0x00000001;

  /* Port B, bit 15 ingång (floating) */
  /* GPIOB_CFGHR nibble för PB15 (bits 31:28): MODE=00, CNF=01 */
  *GPIOB_CFGHR &= ~0xF0000000;
  *GPIOB_CFGHR |= 0x40000000;

  /* PC8 -> EXTI8 (AFIO_EXTICR3, fält EXTI8 = 0x2 för Port C) */
  *AFIO_EXTICR3 &= ~0x0000000F;
  *AFIO_EXTICR3 |= 0x00000002;

  /* EXTI: enable line 8 och välj negativ flank */
  *EXTI_INTENR |= EXTI8_BIT;
  *EXTI_FTENR  |= EXTI8_BIT;
  *EXTI_RTENR  &= ~EXTI8_BIT;

  /* Enable i PFIC */
  *PFIC_IENR1 |= PFIC_EXTI8_IENR1_BIT;
}
```

## Uppgift 7 (6p)
```c
void init_app(void)
{
  /*
    PORT D
      b7-b4 används som utgångar (rader)
      b3-b0 används som ingångar (kolumner)

    Utgångar: open-drain
    Ingångar: pull-up
  */

  /* PD0..PD3: input pull-up  (CNF=10, MODE=00 => 0x8)
     PD4..PD7: output open-drain 10MHz (CNF=01, MODE=01 => 0x5)
     => CFGLR = 0x55558888
  */
  *GPIOD_CFGLR = 0x55558888;

  /* Sätt pull-up för PD0..PD3 (ODR-bitar = 1) och släpp raderna höga */
  *GPIOD_OUTDR |= 0x00FF;
}

unsigned short keyb(void)
{
  unsigned short pad;

  /* Rad 3 (PD7) */
  *GPIOD_BSHR = 0x00F0;         /* alla rader höga */
  *GPIOD_BCR  = (1<<7);         /* driv aktuell rad låg */
  pad  = (unsigned short)(*GPIOD_INDR & 0x000F);

  /* Rad 2 (PD6) */
  *GPIOD_BSHR = 0x00F0;
  *GPIOD_BCR  = (1<<6);
  pad |= (unsigned short)(((*GPIOD_INDR & 0x000F) << 4) & 0x00F0);

  /* Rad 1 (PD5) */
  *GPIOD_BSHR = 0x00F0UL;
  *GPIOD_BCR  = (1<<5);
  pad |= (unsigned short)(((*GPIOD_INDR & 0x000F) << 8) & 0x0F00);

  /* Rad 0 (PD4) */
  *GPIOD_BSHR = 0x00F0UL;
  *GPIOD_BCR  = (1<<4);
  pad |= (unsigned short)(((*GPIOD_INDR & 0x000F) << 12) & 0xF000);

  /* alla rader höga */
  *GPIOD_BSHR = 0x00F0;

  /* returnera inverterat mönster */ 
  return (unsigned short)(pad ^ 0xFFFF); 
}
```

## Uppgift 8 (8p)
```c
void init(void)
{
  /* Port D:
     b6, b5, b2 digital ut (push-pull)
     b1, b0 digital in (pull-up)
     övriga digital in

     CFGLR-nibblar PD0..PD7:
       PD0: input pull-up  => 0x8
       PD1: input pull-up  => 0x8
       PD2: output PP 10MHz=> 0x1
       PD3: input floating => 0x4
       PD4: input floating => 0x4
       PD5: output PP 10MHz=> 0x1
       PD6: output PP 10MHz=> 0x1
       PD7: input floating => 0x4
     => CFGLR = 0x41144188
  */
  *GPIOD_CFGLR = 0x41144188;

  /* Pull-up på b0,b1 */
  *GPIOD_OUTDR |= 0x0003;

  /* Deaktivera styrsignaler (b2,b5,b6) */
  *GPIOD_BCR = (1<<2) | (1<<5) | (1<<6);
}

int open(void)
{
  /* b6=1 */
  *GPIOD_BCR  = (1<<5) | (1<<2);
  *GPIOD_BSHR = (1<<6);

  for(int i=0; i<10; i++)
  {
    if(*GPIOD_INDR & (1<<4))
      return 1; /* Lucka öppnad */
    delay_100ms();
  }
  return 0; /* Kunde inte öppna */
}

int close(void)
{
  /* b5=1 */
  *GPIOD_BCR  = (1<<6) | (1<<2);
  *GPIOD_BSHR = (1<<5);

  for(int i=0; i<20; i++)
  {
    if(*GPIOD_INDR & (1<<3))
      return 1; /* Lucka stängd */
    delay_100ms();
  }
  return 0; /* Kunde inte stänga */
}
```
